cmake_minimum_required(VERSION 3.20)
project(nijilive_unity_compat LANGUAGES CXX)
if(NOT NICXLIVE_SOURCE_DIR)
  message(FATAL_ERROR "Set NICXLIVE_SOURCE_DIR to the nicxlive checkout whose ABI is required")
endif()
find_package(Boost REQUIRED)
add_executable(unity_compat main.cpp)
target_compile_features(unity_compat PRIVATE cxx_std_20)
target_include_directories(unity_compat PRIVATE "${NICXLIVE_SOURCE_DIR}" "${NICXLIVE_SOURCE_DIR}/core" "${NICXLIVE_SOURCE_DIR}/core/third_party" ${Boost_INCLUDE_DIRS})
target_compile_options(unity_compat PRIVATE /utf-8)
